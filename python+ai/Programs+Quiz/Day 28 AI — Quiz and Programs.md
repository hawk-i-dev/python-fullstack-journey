# Day 28 AI — Quiz and Programs

Token usage is more meaningful than request count alone. It tells you how much model work each agent run required. Reported output tokens can include more than visible text, so use provider usage data—not character estimates—for tracking. [Official OpenAI token-counting guide](https://developers.openai.com/api/docs/guides/token-counting)

## Program 1 — Test an agent-run record

Create `tests/test_observability.py`:

```python
from app.observability import record_completed_agent_run


def test_completed_agent_run_is_recorded(db):
    agent_run = record_completed_agent_run(
        db=db,
        user_id="alice",
        conversation_id="conversation-123",
        model="demo-model",
        input_tokens=120,
        output_tokens=30,
        duration_ms=850,
    )

    assert agent_run.user_id == "alice"
    assert agent_run.input_tokens == 120
    assert agent_run.output_tokens == 30
    assert agent_run.total_tokens == 150
    assert agent_run.duration_ms == 850
    assert agent_run.status == "completed"
```

## Program 2 — Create a user-summary helper

Add this to `app/observability.py`:

```python
from sqlalchemy import func, select

from app.models import AgentRun


def get_user_run_summary(
    db: Session,
    user_id: str,
) -> dict[str, int]:
    statement = select(
        func.count(AgentRun.id),
        func.coalesce(func.sum(AgentRun.input_tokens), 0),
        func.coalesce(func.sum(AgentRun.output_tokens), 0),
        func.coalesce(func.sum(AgentRun.total_tokens), 0),
    ).where(
        AgentRun.user_id == user_id,
        AgentRun.status == "completed",
    )

    total_runs, input_tokens, output_tokens, total_tokens = (
        db.execute(statement).one()
    )

    return {
        "total_runs": int(total_runs),
        "input_tokens": int(input_tokens),
        "output_tokens": int(output_tokens),
        "total_tokens": int(total_tokens),
    }
```

Test it:

```python
from app.observability import (
    get_user_run_summary,
    record_completed_agent_run,
)


def test_summary_only_includes_selected_user(db):
    record_completed_agent_run(
        db=db,
        user_id="alice",
        conversation_id="conversation-1",
        model="demo-model",
        input_tokens=100,
        output_tokens=20,
        duration_ms=500,
    )

    record_completed_agent_run(
        db=db,
        user_id="bob",
        conversation_id="conversation-2",
        model="demo-model",
        input_tokens=500,
        output_tokens=100,
        duration_ms=900,
    )

    summary = get_user_run_summary(
        db=db,
        user_id="alice",
    )

    assert summary["total_runs"] == 1
    assert summary["input_tokens"] == 100
    assert summary["output_tokens"] == 20
    assert summary["total_tokens"] == 120
```

## Program 3 — Display run metrics in React

Add this type in `frontend/src/api.ts`:

```typescript
export type UsageMetrics = {
  total_runs: number;
  input_tokens: number;
  output_tokens: number;
  total_tokens: number;
  average_duration_ms: number;
};
```

Add an API function:

```typescript
export async function getUsageMetrics(
  apiToken: string,
): Promise<UsageMetrics> {
  const response = await fetch(
    `${API_URL}/usage/me/summary`,
    {
      headers: {
        Authorization: `Bearer ${apiToken}`,
      },
    },
  );

  if (!response.ok) {
    throw new Error("Could not load usage metrics");
  }

  return response.json() as Promise<UsageMetrics>;
}
```

Display in `App.tsx`:

```tsx
{metrics && (
  <section>
    <h2>Your AI usage</h2>
    <p>Completed runs: {metrics.total_runs}</p>
    <p>Total tokens: {metrics.total_tokens}</p>
    <p>Input tokens: {metrics.input_tokens}</p>
    <p>Output tokens: {metrics.output_tokens}</p>
    <p>
      Average response time:{" "}
      {metrics.average_duration_ms} ms
    </p>
  </section>
)}
```
